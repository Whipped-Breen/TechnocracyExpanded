TNA_congress_to_continental = {
	set_temp_variable = { temp_senators_support = senators_support }
	divide_temp_variable = { temp_senators_support = senators_total }
	multiply_temp_variable = { temp_senators_support = continental_total }
	set_variable = { var = continental_support value = temp_senators_support }
	round_variable = continental_support
	TNA_recalc_administration_icon_frames = yes
}

TNA_continental_get_quorum = {
	set_temp_variable = { temp_quorum = continental_total }
	divide_temp_variable = { temp_quorum = 3 }
	multiply_temp_variable = { temp_quorum = 2 }
	set_variable = { var = continental_quorum value = temp_quorum }
	round_variable = continental_quorum
}

TNA_recalc_administration_icon_frames = {
	
	set_temp_variable = { continental_temp = continental_support }
	divide_temp_variable = { continental_temp = continental_total }
	multiply_temp_variable = { continental_temp = 21 }
	if = {
		limit = { check_variable = { continental_temp > 0 } }
		clamp_temp_variable = {
			var = continental_temp
			min = 1
			max = 21
		}
	}
	else = {
		set_temp_variable = { continental_temp = 1 }
	}
	set_variable = { current_continental_icon_frame = continental_temp }
}

TNA_canada_support_time = {
	set_variable = { tni_drf = can.TNA_tpc_neutrality_drift }
	add_to_variable = { tni_drf = 0.03 }

	set_temp_variable = { tni_cmt = 35 }
	multiply_temp_variable = { tni_cmt = tni_drf }

	set_variable = { check_popularity = 50 }
	subtract_from_variable = { check_popularity = tni_cmt }

	set_variable = { tech_popularity = party_popularity_100@neutrality }

	set_variable = { growth_days = check_popularity }
	subtract_from_variable = { growth_days = tech_popularity }
	divide_variable = { growth_days = tni_drf }
	round_variable = growth_days

	if = {
		limit = { check_variable = { growth_days < 0 } }
		set_variable = { growth_days = 0 }
	}
}

TNA_join_research_north_american_technate = {
	if = {
		limit = {
			USA = {
				NOT = { has_cosmetic_tag = TNA_north_american_technate }
			}
		}
		USA = {
			if = {
				limit = {
					NOT = { is_in_tech_sharing_group = TNA_north_american_technate }
				}
				add_to_tech_sharing_group = TNA_north_american_technate
			}
		}
		add_to_tech_sharing_group = TNA_north_american_technate
	}
}
TNA_end_research_north_american_technate = {

			is_in_tech_sharing_group = TNA_north_american_technate
		}
		remove_from_tech_sharing_group = TNA_north_american_technate
	}
}
